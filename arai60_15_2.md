class Solution:
    def firstUniqChar(self, s: str) -> int:
        character_to_count = {}
        for character in s:
            if character not in character_to_count:
                character_to_count[character] = 0
            character_to_count[character] += 1
        
        for index, character in enumerate(s):
            if character_to_count[character] == 1:
                return index
        
        return -1
```

class Solution:
    def firstUniqChar(self, s: str) -> int:
        already_appeared = set()
        character_to_count = defaultdict(int)
        for character in s:
            if character in already_appeared and character in character_to_count:
                del character_to_count[character]
            elif character in already_appeared:
                continue
            else:
                character_to_count[character] = 1
                already_appeared.add(character)

        first_unique = next(iter(character_to_count.keys()))
        for index, character in enumerate(s):
            if character == first_unique:
                return index
        
        return -1
```
