```python
class Solution:
    def firstUniqChar(self, s: str) -> int:
        character_to_count = {}
        for character in s:
            if character not in character_to_count:
                character_to_count[character] = 0
            character_to_count[character] += 1
        
        for index, character in enumerate(s):
            if character_to_count[character] == 1:
                return index
        
        return -1
```
