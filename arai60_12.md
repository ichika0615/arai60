# Step 1 
## ランレンクスエンコーディングでまとめる。文字列をランレンクスエンコードして、それをキーとする。バリューには対応する文字列を格納する。

```python
class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        def run_length_encoding(word):
            character_to_count = {}
            for word_item in word:
                character_to_count.setdefault(word_item, 0)
                character_to_count[word_item] += 1
            sorted_character_to_count = sorted(character_to_count.items())
            run_length = ''
            for character, count in sorted_character_to_count:
                run_length = run_length + character + str(count)
            return run_length

        
        group = defaultdict(list)
        for strs_item in strs:
            group[run_length_encoding(strs_item)].append(strs_item)
        
        return [words for _, words in group.items()]
```

## ハッシュ値をキーにする。バリューには対応する文字列を格納する。
```python
class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        def convert_to_hash(word):
            hash_value = 0
            for word_item in word:
                hash_value += ord(word_item)**4 - ord('a')*2 + 1
            return hash_value
        
        hash_to_words = defaultdict(list)
        for strs_item in strs:
            hash_to_words[convert_to_hash(strs_item)].append(strs_item)
        
        return list(hash_to_words.values()))
```

他の方の解法も見てみる。
## キーとして、アルファベットの出現頻度をタプルを用いる。
```python
class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        def characters_frequency(word):
            codepoints = 26
            character_to_frequency = [0] * codepoints
            for word_item in word:
                character_to_frequency[ord(word_item) - ord('a')] += 1
            return tuple(character_to_frequency)
        
        frequency_to_words = defaultdict(list)
        for strs_item in strs:
            frequency_to_words[characters_frequency(strs_item)].append(strs_item)
        
        return list(frequency_to_words.values())
```
## ソートした文字列をキーとする。
```python
class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        def sort_string(string):
            string_list = list(string)
            string_list.sort()
            return ''.join(string_list)

        sorted_to_words = defaultdict(list)
        for strs_item in strs:
            sorted_to_words[sort_string(strs_item)].append(strs_item)
        
        return list(sorted_to_words.values())
```
