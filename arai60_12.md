## ランレンクスエンコーディングでまとめる。
```python
class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        def run_length_encoding(word):
            character_to_count = {}
            for word_item in word:
                character_to_count.setdefault(word_item, 0)
                character_to_count[word_item] += 1
            sorted_character_to_count = sorted(character_to_count.items())
            run_length = ''
            for character, count in sorted_character_to_count:
                run_length = run_length + character + str(count)
            return run_length

        
        group = defaultdict(list)
        for strs_item in strs:
            group[run_length_encoding(strs_item)].append(strs_item)
        
        return [words for _, words in group.items()]
```
