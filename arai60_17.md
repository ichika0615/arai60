## BFS
```python
class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        def remove_islands(colum, row):
            lands = deque([(colum, row)])
            while lands:
                column, row = lands.popleft()
                if not (0 <= column < column_length and 0 <= row < row_length):
                    continue 
                if grid[column][row] == '0':
                    continue
                grid[column][row] = '0'
                directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]
                for di_c, di_r in directions:
                    lands.append((column + di_c, row + di_r))
        
        column_length = len(grid)
        row_length = len(grid[0])
        number_of_islands = 0
        for i in range(column_length):
            for j in range(row_length):
                if grid[i][j] == '1':
                    number_of_islands += 1
                    remove_islands(i, j)
        
        return number_of_islands
```

## Union Find

- ユニオンファインドを用いる。二次元のマップを一次元化してlands.parentsにて管理することになる。陸地を検出したらunionしてまとめて、結局繋がってる陸地はグルーピングされるから島の数がわかる。

class UnionFind:
    def __init__(self, size):
        self.parents = [i for i in range(size)]

    def find(self, node):
        root = node
        while self.parents[root] != root:
            root = self.parents[root]
        return root
    
    def union(self, node_1, node_2):
        root_1 = self.find(node_1)
        root_2 = self.find(node_2)
        if root_1 != root_2:
            self.parents[root_2] = root_1
    
class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        height = len(grid)
        width = len(grid[0])
        lands = UnionFind(height*width)

        def get_index_in_unionfind(row, column):
            return row * width + column
        
        directions = [(0, 1), (1, 0), (0, -1), (-1, 0)]
        for h in range(height):
            for w in range(width):
                if grid[h][w] == '1':
                    for dir_r, dir_c in directions:
                        if 0 <= h + dir_r < height and 0 <= w + dir_c < width and grid[h + dir_r][w + dir_c] == '1':
                            lands.union(get_index_in_unionfind(h, w), get_index_in_unionfind(h + dir_r, w + dir_c))
        
        types_of_islands = set()
        for h in range(height):
            for w in range(width):
                if grid[h][w] == '1':
                    types_of_islands.add(lands.parents[get_index_in_unionfind(h, w)])
        
        return len(types_of_islands)

## Union Find
- parents を辞書型で定義している人がいて、そちらの方がわかりやすく感じる。
```python
WATER = '0'
LAND = '1'

class UnionFind:
    def __init__(self, islands):
        self.parents = {area: area for area in islands}

    def find(self, x):
        while self.parents[x] != x:
            x = self.parents[x]
        return x
    
    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root != y_root:
            self.parents[x_root] = y_root
    
class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        row_length = len(grid)
        column_length = len(grid[0])
        islands = []

        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND:
                    islands.append((row, column))
        
        islands_uf = UnionFind(islands)
        directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]

        for row, column in islands_uf.parents:
            for d_row, d_column in directions:
                if (row + d_row, column + d_column) in islands_uf.parents:
                    islands_uf.union((row, column), (row + d_row, column + d_column))
        
        roots = set()

        for area in islands_uf.parents:
            roots.add(islands_uf.find(area))
        
        return len(roots)

```
