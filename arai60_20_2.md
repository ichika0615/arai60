class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        words_set = set(wordList)
        if endWord not in words_set:
            return 0

        words_to_visit = deque([(beginWord, 1)])
        seen_words = set([beginWord])

        def get_next_words(word):
            next_words = []
            for i in range(len(word)):
                for c in string.ascii_lowercase:
                    new_word = ''.join([word[:i], c, word[i + 1:]])
                    if new_word in words_set and new_word not in seen_words:
                        next_words.append(new_word)
                        seen_words.add(new_word)
            return next_words
        
        while words_to_visit:
            word, depth = words_to_visit.popleft()
            if word == endWord:
                return depth
            for next_word in get_next_words(word):
                words_to_visit.append((next_word, depth + 1))
        
        return 0
