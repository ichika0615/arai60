## BFSでノードを上から探索する。葉っぱまで来たらその道の合計をセットに記録する。最後にtargetSumがそのセットの中にあるかをみる。
```python
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        if not root:
            return False
        
        sums_found = set()
        nodes_sum_pair = deque([(root, root.val)])
        while nodes_sum_pair:
            node, sum_so_far = nodes_sum_pair.popleft()
            if not node.left and not node.right:
                sums_found.add(sum_so_far)
            if node.left:
                nodes_sum_pair.append((node.left, sum_so_far + node.left.val))
            if node.right:
                nodes_sum_pair.append((node.right, sum_so_far + node.right.val))
        
        return targetSum in sums_found
```

## 再帰
- targetSumを変えていって再帰
```python
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        if not root:
            return False
        
        if not root.left and not root.right and root.val == targetSum:
            return True
        
        return self.hasPathSum(root.left, targetSum - root.val) or self.hasPathSum(root.right, targetSum - root.val)
```

```python
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        def has_path_sum_helper(node, current_sum):
            if not node:
                return False
            
            current_sum = current_sum + node.val
            if not node.left and not node.right and current_sum == targetSum:
                return True
            
            return has_path_sum_helper(node.left, current_sum) or has_path_sum_helper(node.right, current_sum)
        
        return has_path_sum_helper(root, 0)
```

```python
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        if not root:
            return False
        
        def is_leaf(node):
            return not node.left and not node.right
        
        node_and_current_sum = [(root, root.val)]
        while node_and_current_sum:
            node, current_sum = node_and_current_sum.pop()
            if is_leaf(node) and current_sum == targetSum:
                return True
            if node.left:
                node_and_current_sum.append((node.left, current_sum + node.left.val))
            if node.right:
                node_and_current_sum.append((node.right, current_sum + node.right.val))
        
        return False
```
