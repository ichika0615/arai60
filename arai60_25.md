## BFSでノードを上から探索する。葉っぱまで来たらその道の合計をセットに記録する。最後にtargetSumがそのセットの中にあるかをみる。
```python
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        if not root:
            return False
        
        sums_found = set()
        nodes_sum_pair = deque([(root, root.val)])
        while nodes_sum_pair:
            node, sum_so_far = nodes_sum_pair.popleft()
            if not node.left and not node.right:
                sums_found.add(sum_so_far)
            if node.left:
                nodes_sum_pair.append((node.left, sum_so_far + node.left.val))
            if node.right:
                nodes_sum_pair.append((node.right, sum_so_far + node.right.val))
        
        return targetSum in sums_found
```

```python
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        if not root:
            return False
        
        nodes_sum_pair = deque([(root, root.val)])
        while nodes_sum_pair:
            node, sum_so_far = nodes_sum_pair.popleft()
            if not node.left and not node.right:
                if targetSum == sum_so_far:
                    return True
            if node.left:
                nodes_sum_pair.append((node.left, sum_so_far + node.left.val))
            if node.right:
                nodes_sum_pair.append((node.right, sum_so_far + node.right.val))
        
        return False
```
- DFSは葉っぱにたどり着くのは早いが、上に戻っていくことがあるので合計値を引いたり足したりするのが面倒くさい。それに、BFSと比べて目に見えて早いとも思えない。
