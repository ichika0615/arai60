# Step 2
## 再帰DFSを用いる
```python
LAND = 1
WATER = 0
MAX_RECURSION = 10**4
sys.setrecursionlimit(MAX_RECURSION)

class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        if not grid:
            return 0

        row_length = len(grid)
        column_length = len(grid[0])
        is_visited = [[False] * column_length for _ in range(row_length)]

        def is_in_grid(row, column):
            return 0 <= row < row_length and 0 <= column < column_length

        def traverse_and_measure_island(row, column):
            size = 0
            if not is_in_grid(row, column):
                return size
            if is_visited[row][column]:
                return size
            if grid[row][column] == WATER:
                return size
            is_visited[row][column] = True
            size += 1
            return (
                    size + traverse_and_measure_island(row + 1, column)
                    + traverse_and_measure_island(row - 1, column)
                    + traverse_and_measure_island(row, column + 1)
                    + traverse_and_measure_island(row, column - 1)
            )
            
        max_size = 0

        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND and not is_visited[row][column]:
                    max_size = (max_size, traverse_and_measure_island(row, column))

        return max_size

```
- 再帰深さは最大50**2
- 時間計算量: O(nm) (Pythonだから実行時間は2500/100万second くらい？)
- 空間計算量: O(nm) (is_visitedとコールスタックはいずれもO(nm))
