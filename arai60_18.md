# Step 1
## DFSで探索して、探索するごとに`island_size`をインクリメントしていって、size_of_islandsに格納していき、その最大値を出力する。
```python
LAND = '1'
WATER = '0'

class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        def is_in_grid(row, column):
            return 0 <= row < row_length and 0 <= column < column_length

        def visit_island(row, column, island_size) -> int:
            if not is_in_grid(row, column):
                return island_size
            if grid[row][column] == WATER or is_visited[row][column]:
                return island_size
            
            is_visited[row][column] = True
            island_size += 1
            visit_island(row + 1, column, island_size)
            visit_island(row - 1, column, island_size)
            visit_island(row, column + 1, island_size)
            visit_island(row, column - 1, island_size)

        row_length = len(grid)
        column_length = len(grid[0])
        is_visited = [[False] * column_length for _ in range(row_length)]
        size_of_islands = set()

        if not grid:
            return 0
        
        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND and not is_visited[row][column]:
                    size_of_islands.add(visit_island(row, column, 0))

        if not size_of_islands:
            return 0
        return max(size_of_islands)
```
