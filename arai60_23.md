```python
        if not root1 and not root2:
```

```python
class Solution:
    def mergeTrees(self, root1: Optional[TreeNode], root2: Optional[TreeNode]) -> Optional[TreeNode]:
        if not root1 and not root2:
            return None
        if not root1:
            return root2
        if not root2:
            return root1

        def get_value(node):
            if not node:
                return 0
            else:
                return node.val

        merged = TreeNode()
        given_nodes_merged_node = [(root1, root2, merged)]  # This is a stack.
        while given_nodes_merged_node:
            node1, node2, merged_node = given_nodes_merged_node.pop()
            merged_node.val = get_value(node1) + get_value(node2)
            if (node1 and node1.left) or (node2 and node2.left):
                merged_node.left = TreeNode()
                given_nodes_merged_node.append((
                    node1.left if node1 else None,
                    node2.left if node2 else None,
                    merged_node.left
                ))
            if (node1 and node1.right) or (node2 and node2.right):
                merged_node.right = TreeNode()
                given_nodes_merged_node.append((
                    node1.right if node1 else None,
                    node2.right if node2 else None,
                    merged_node.right
                ))


        return merged
```

```python
class Solution:
    def mergeTrees(self, root1: Optional[TreeNode], root2: Optional[TreeNode]) -> Optional[TreeNode]:
        if not root1 and not root2:
            return None
        if not root1:
            return root2
        if not root2:
            return root1

        node1_node2_pair = deque([(root1, root2)])
        while node1_node2_pair:
            node1, node2 = node1_node2_pair.popleft()
            if not node2:
                continue
            node1.val += node2.val
            if node1.left:
                node1_node2_pair.append((node1.left, node2.left))
            if node1.right:
                node1_node2_pair.append((node1.right, node2.right))
            if not node1.left and node2.left:
                node1.left = TreeNode(0)
                node1_node2_pair.append((node1.left, node2.left))
            if not node1.right and node2.right:
                node1.right = TreeNode(0)
                node1_node2_pair.append((node1.right, node2.right))
        
        return root1
```
