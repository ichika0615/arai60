class Solution:
    def lengthOfLIS(self, nums: List[int]) -> int:
        increasing_subsequences = []
        is_seen = [False] * len(nums)
        for i in range(len(nums)):
            if is_seen[i]:
                continue
            increasing_sequence = [nums[i]]
            for j in range(i + 1, len(nums)):
                if nums[j] > increasing_sequence[-1]:
                    increasing_sequence.append(nums[j])
                    is_seen[j] = True
            increasing_subsequences.append((increasing_sequence, len(increasing_sequence)))

        max_length = 0
        for _, length in increasing_subsequences:
            if max_length < length:
                max_length = length
        
        return max_length

class Solution:
    def lengthOfLIS(self, nums: List[int]) -> int:
        if not nums:
            return 0

        length_nums = len(nums)
        nums_to_length_until_it = [1] * length_nums

        for i in range(length_nums):
            for j in range(i):
                if nums[i] > nums[j]:
                    nums_to_length_until_it[i] = max(nums_to_length_until_it[i], nums_to_length_until_it[j] + 1)

        return max(nums_to_length_until_it)
