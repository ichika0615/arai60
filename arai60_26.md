## BFS で1レベルずつ格納していく。
```python
class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:
            return []
        
        def convert_nodes_to_values(nodes_list):
            values_list = []
            for node in nodes_list:
                values_list.append(node.val)
            return values_list
                
        current_level_nodes = [root]
        values_per_levels = []
        while current_level_nodes:
            values_per_levels.append(convert_nodes_to_values(current_level_nodes))
            next_level_nodes = []
            for node in current_level_nodes:
                if node.left:
                    next_level_nodes.append(node.left)
                if node.right:
                    next_level_nodes.append(node.right)
            current_level_nodes = next_level_nodes
        
        return values_per_levels
```

```python
class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:
            return []

        level_to_node_values = defaultdict(list)
        
        nodes_and_depth = deque([(root, 1)])
        while nodes_and_depth:
            node, depth = nodes_and_depth.popleft()
            level_to_node_values[depth].append(node.val)
            if node.left:
                nodes_and_depth.append((node.left, depth + 1))
            if node.right:
                nodes_and_depth.append((node.right, depth + 1))
        
        nodes_per_level = []
        for level, node_values in level_to_node_values.items():
            nodes_per_level.append(node_values)
        
        return nodes_per_level
```
