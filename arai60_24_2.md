```python
class Solution:

        def sorted_array_to_BST_helper(left_index, right_index) -> Optional[TreeNode]:
            if left_index > right_index:
                return None
            middle_index = (left_index + right_index) // 2
            node = TreeNode(nums[middle_index])
            
            node.left = sorted_array_to_BST_helper(left_index, middle_index - 1)
            node.right = sorted_array_to_BST_helper(middle_index + 1, right_index)
            
            return node
        
        return sorted_array_to_BST_helper(0, len(nums) - 1)
```

```python
class Solution:
    def sortedArrayToBST(self, nums: List[int]) -> Optional[TreeNode]:
        if not nums:
            return None
        
        root = TreeNode()
        nodes_left_right = [(root, 0, len(nums) - 1)]
        
        while nodes_left_right:
            node, left_index, right_index = nodes_left_right.pop()
            middle_index = (left_index + right_index) // 2
            node.val = nums[middle_index]
            
            if left_index < middle_index:
                node.left = TreeNode()
                nodes_left_right.append((node.left, left_index, middle_index - 1))
            if middle_index < right_index:
                node.right = TreeNode()
                nodes_left_right.append((node.right, middle_index + 1, right_index))
        
        return root
```

```python
class Solution:
    def sortedArrayToBST(self, nums: List[int]) -> Optional[TreeNode]:
        if not nums:
            return None
        middle_index = (0 + len(nums) - 1) // 2
        root = TreeNode(nums[middle_index])
        
        nodes_left_right = [(root, 0, len(nums) - 1)]
        while nodes_left_right:
            node, left_index, right_index = nodes_left_right.pop()
            middle_index = (left_index + right_index) // 2
            
            if left_index < middle_index:
                left_middle_index = (left_index + middle_index - 1) // 2
                node.left = TreeNode(nums[left_middle_index])
                nodes_left_right.append((node.left, left_index, middle_index - 1))
            
            if middle_index < right_index:
                right_middle_index = (middle_index + 1 + right_index) // 2
                node.right = TreeNode(nums[right_middle_index])
                nodes_left_right.append((node.right, middle_index + 1, right_index))
        
        return root
```
