# Step 1
- BFSを使えば最短距離を導けると思った。
- 木構造があることは認識できる。どうやってBFSを実行するか。
```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        if endWord not in wordList:
            return 0

        def is_different_by_single_char(word_1, word_2) -> bool:
            if len(word_1) != len(word_2):
               return False
            num_difference = 0
            for i in range(len(word_1)):
                if word_1[i] != word_2[i]:

                    num_difference += 1
            return num_difference == 1

        word_to_accessible_words = defaultdict(list)
        for word_1 in wordList:
            for word_2 in wordList:
                if is_different_by_single_char(word_1, word_2):
                    word_to_accessible_words[word_1].append(word_2)
        
        words_to_visit = deque([])
        for word in wordList:
            if is_different_by_single_char(beginWord, word):
                words_to_visit.append((word, 2))
        while words_to_visit:
            present_word, present_depth = words_to_visit.popleft()
            if present_word == endWord:
                return present_depth
            for word in word_to_accessible_words[present_word]:
                words_to_visit.append((word, present_depth + 1))
        
        return 0
```
- TLE (17/51 testcases passed)
- 木構造を明確にしようとして、`word_to_accessible_words`の構築にO(n^2)かけている。

## setに直してから実行。
```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        if endWord not in wordList:
            return 0
        
        wordList = set(wordList)
        seen_words = set()
        if beginWord in wordList:
            seen_words.add(beginWord)
        words_to_visit = deque([(beginWord, 1)])
        while words_to_visit:
            present_word, present_depth = words_to_visit.popleft()
            if present_word == endWord:
                return present_depth
            for i in range(len(present_word)):
                for alphabet in string.ascii_lowercase:
                    new_word = present_word[:i] + alphabet + present_word[i + 1:]
                    if new_word in wordList and new_word not in seen_words:
                        words_to_visit.append((new_word, present_depth + 1))
                        seen_words.add(new_word)
        
        return 0     
```
- 時間計算量: O(NK) (Nは要素の数、Kは各要素の長さ)
- 空間計算量: O(N)
- うまく動いた。`if A in Array` は、listでは線形探索で見ていくのでO(n)かかるが、setならハッシュで見ていくので大きさに関係なくO(1)
- setにオブジェクトを追加するときは、そのハッシュを計算して対応する場所に記憶する。存在チェックは候補をハッシュ計算してそこにオブジェクトがあるかどうかを見る(大きさによらず一定の時間でできる)。setに順序がない理由でもある。
- https://stackoverflow.com/questions/8929284/what-makes-sets-faster-than-lists

## keyを用いて検索できるようにする。
```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        key_to_words = defaultdict(list)
        word_to_keys = defaultdict(list)
        for i in range(len(beginWord)):
            key = (beginWord[:i], beginWord[i+1:])
            word_to_keys[beginWord].append(key)
            key_to_words[key].append(beginWord)

        for word in wordList:
            for i in range(len(word)):
                key = (word[:i], word[i + 1:])
                word_to_keys[word].append(key)
                key_to_words[key].append(word)
                
        words_to_visit = deque([(beginWord, 1)])
        seen_words = set([beginWord])
        while words_to_visit:
            present_word, present_depth = words_to_visit.popleft()
            if present_word == endWord:
                return present_depth
            keys = word_to_keys[present_word]
            for key in keys:
                for word in key_to_words[key]:
                    if word not in seen_words:
                        words_to_visit.append((word, present_depth + 1))
                        seen_words.add(word)
                    
        return 0       
```

```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        wordList = set(wordList)
        if endWord not in wordList:
            return 0
        words_to_visit = deque([(beginWord, 1)])
        seen_words = set()
        
        while words_to_visit:
            present_word, present_depth = words_to_visit.popleft()
            if present_word == endWord:
                return present_depth
            for i in range(len(present_word)):
                for c in string.ascii_lowercase:
                    new_word = list(present_word[:i]) # join to get str later
                    new_word.append(c)
                    new_word.extend(list(present_word[i+1:]))
                    new_word = ''.join(new_word)
                    if new_word in wordList and new_word not in seen_words:
                        words_to_visit.append((new_word, present_depth + 1))
                        seen_words.add(new_word)
        
        return 0
```
- これlist型にするのに線形で舐めるし諸々オーバーヘッドがかかってそうで、大人しくstr型で更新して良さそう。
