# Step 1
- BFSを使えば最短距離を導けると思った。
- 木構造があることは認識できる。どうやってBFSを実行するか。
```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        if endWord not in wordList:
            return 0

        def is_different_by_single_char(word_1, word_2) -> bool:

            if len(word_1) != len(word_2):
               return False
            num_difference = 0
            for i in range(len(word_1)):
                if word_1[i] != word_2[i]:

                    num_difference += 1
            return num_difference == 1

        word_to_accessible_words = defaultdict(list)
        for word_1 in wordList:
            for word_2 in wordList:
                if is_different_by_single_char(word_1, word_2):
                    word_to_accessible_words[word_1].append(word_2)
        
        words_to_visit = deque([])
        for word in wordList:
            if is_different_by_single_char(beginWord, word):
                words_to_visit.append((word, 2))
        while words_to_visit:
            present_word, present_depth = words_to_visit.popleft()
            if present_word == endWord:
                return present_depth
            for word in word_to_accessible_words[present_word]:
                words_to_visit.append((word, present_depth + 1))
        
        return 0
```
- TLE (17/51 testcases passed)
- 木構造を明確にしようとして、`word_to_accessible_words`の構築にO(n^2)かけている。
