```python
LAND = '1'
WATER = '0'
MAX_RECURSION = 10**5
sys.setrecursionlimit(MAX_RECURSION)

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        def is_in_grid(row, column):
            return 0 <= row < row_length and 0 <= column < column_length

        def traverse_island(row, column):
            if not is_in_grid(row, column):
                return
            if grid[row][column] == WATER:
                return
            if is_visited[row][column]:
                return
            
            is_visited[row][column] = True
            traverse_island(row + 1, column)
            traverse_island(row - 1, column)
            traverse_island(row, column + 1)
            traverse_island(row, column - 1)
            
        if not grid:
            raise ValueError('invalid input')
            # 例外処理はどこまで一生懸命やるべきかな？？

        row_length = len(grid)
        column_length = len(grid[0])
        is_visited = [[False] * column_length for _ in range(row_length)]
        number_of_islands = 0

        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND and not is_visited[row][column]:
                    traverse_island(row, column)
                    number_of_islands += 1
    
        return number_of_islands
```

# 非再帰DFS
```python
LAND = '1'
WATER = '0'

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        if not grid:
          return 0

        def is_in_grid(row, column):
            return 0 <= row < row_length and 0 <= column < column_length
                
        row_length = len(grid)
        column_length = len(grid[0])
        is_visited = [[False] * column_length for _ in range(row_length)]
        directions = [(1, 0), (-1, 0), (0, 1), (0, -1)]
        number_of_islands = 0
       
        def traverse_island(row, column):
            lands_to_visit = [(row, column)]
            while lands_to_visit:
                present_row, present_column = lands_to_visit.pop()
                if not is_in_grid(present_row, present_column):
                    continue
                if grid[present_row][present_column] == WATER:
                    continue
                if is_visited[present_row][present_column]:
                    continue
                is_visited[present_row][present_column] = True
                for dir_row, dir_column in directions:
                    new_row = present_row + dir_row
                    new_column = present_column + dir_column
                    lands_to_visit.append((new_row, new_column))
                
        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND and not is_visited[row][column]:
                    traverse_island(row, column)
                    number_of_islands += 1

        return number_of_islands
```
