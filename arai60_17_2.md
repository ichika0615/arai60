# 再帰DFS
```python
LAND = '1'
WATER = '0'
MAX_RECURSION = 10**5
sys.setrecursionlimit(MAX_RECURSION)

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        def is_in_grid(row, column):
            return 0 <= row < row_length and 0 <= column < column_length

        def traverse_island(row, column):
            if not is_in_grid(row, column):
                return
            if grid[row][column] == WATER:
                return
            if is_visited[row][column]:
                return
            
            is_visited[row][column] = True
            traverse_island(row + 1, column)
            traverse_island(row - 1, column)
            traverse_island(row, column + 1)
            traverse_island(row, column - 1)
            
        if not grid:
            raise ValueError('invalid input')
            # 例外処理はどこまで一生懸命やるべきかな？？

        row_length = len(grid)
        column_length = len(grid[0])
        is_visited = [[False] * column_length for _ in range(row_length)]
        number_of_islands = 0

        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND and not is_visited[row][column]:
                    traverse_island(row, column)
                    number_of_islands += 1
    
        return number_of_islands
```

# 非再帰DFS
```python
LAND = '1'
WATER = '0'

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        if not grid:
          return 0

        def is_in_grid(row, column):
            return 0 <= row < row_length and 0 <= column < column_length
                
        row_length = len(grid)
        column_length = len(grid[0])
        is_visited = [[False] * column_length for _ in range(row_length)]
        directions = [(1, 0), (-1, 0), (0, 1), (0, -1)]
        number_of_islands = 0
       
        def traverse_island(row, column):
            lands_to_visit = [(row, column)]
            while lands_to_visit:
                present_row, present_column = lands_to_visit.pop()
                if not is_in_grid(present_row, present_column):
                    continue
                if grid[present_row][present_column] == WATER:
                    continue
                if is_visited[present_row][present_column]:
                    continue
                is_visited[present_row][present_column] = True
                for dir_row, dir_column in directions:
                    new_row = present_row + dir_row
                    new_column = present_column + dir_column
                    lands_to_visit.append((new_row, new_column))
                
        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND and not is_visited[row][column]:
                    traverse_island(row, column)
                    number_of_islands += 1

        return number_of_islands
```
# BFS

# UnionFind
## とりあえず一番最初に書こうとしたもの。
```python
LAND = '1'
WATER = '0'

class UnionFind:
    def __init__(self, size):
        self.parents = [i for i in range(size)]

    def find(self, element):
        while element != self.parents[element]:
            element = self.parents[element]
        return element

    def union(self, el_1, el_2):
        root_1 = self.find(el_1)
        root_2 = self.find(el_2)
        if root_1 != root_2:
            self.parents[root_2] = root_1

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        if not grid:
            raise ValueError('invalid input')

        row_length = len(grid)
        column_length = len(grid[0])
        directions = [(1, 0), (0, 1)]
        uf = UnionFind(row_length * column_length)

        def is_in_grid(row, column):
            return 0 <= row < row_length and 0 <= column < column_length
        
        def get_uf_index(row, column):
            return row * column_length + column
            
        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND:
                    for dir_row, dir_column in directions:
                        new_row = row + dir_row
                        new_column = column + dir_column
                        if is_in_grid(new_row, new_column) and grid[new_row][new_column] == LAND:
                            uf.union(get_uf_index(row, column), get_uf_index(new_row, new_column))
        
        roots = set()
        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND:
                    roots.add(uf.find(get_uf_index(row, column)))
        
        return len(roots)
```

## 経路圧縮を施した
```python
LAND = '1'

class UnionFind:
    def __init__(self, size):
        self.parents = [i for i in range(size)]

    def find(self, element):
        if self.parents[element] != element:
            self.parents[element] = self.find(self.parents[element])
        return self.parents[element]

    def union(self, el_1, el_2):
        root_1 = self.find(el_1)
        root_2 = self.find(el_2)
        if root_2 != root_1:
            self.parents[root_2] = root_1

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        if not grid:
            raise ValueError('invalid input')
        if not isinstance(grid, list):
            raise TypeError('invalid input')

        row_length = len(grid)
        column_length = len(grid[0])
        uf = UnionFind(row_length * column_length)
        directions = [(1, 0), (0, 1)]
        def is_in_grid(row, column):
            return 0 <= row < row_length and 0 <= column < column_length

        def get_uf_index(row, column):
            return row * column_length + column
        
        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND:
                    for dir_row, dir_column in directions:
                        new_row = row + dir_row
                        new_column = column + dir_column
                        if is_in_grid(new_row, new_column) and grid[new_row][new_column] == LAND:
                            uf.union(get_uf_index(row, column), get_uf_index(new_row, new_column))
        
        roots = set()
        for row in range(row_length):
            for column in range(column_length):
                if grid[row][column] == LAND:
                    roots.add(uf.find(get_uf_index(row, column)))
        
        return len(roots)
```
経路圧縮により、findの時間計算量が改善される。もし親子が一列に並ぶと、最悪計算量は$O(N)$となるが、経路圧縮により木が平らに維持され、計算量は$O(α(N))$となる。$α$は逆アッカーマン。
